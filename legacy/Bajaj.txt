function generateCustomerCode() {
    const L = 'ABCDEFGHJKLMNPQRSTUVWXYZ', D = '0123456789';
    let code;
    do {
        code = '';
        for (let i = 0; i < 3; i++) code += L[Math.floor(Math.random() * L.length)];
        for (let i = 0; i < 3; i++) code += D[Math.floor(Math.random() * D.length)];
    } while (appData.customers.some(c => c.password === code));
    return code;
}

function setCloudStatus(status, text) {
    const el = document.getElementById('cloudStatus');
    if (!el) return;
    el.className = 'cloud-status cloud-' + status;
    const ic = status === 'online' ? 'fa-cloud' : status === 'syncing' ? 'fa-cloud-arrow-up fa-spin' : 'fa-cloud-slash';
    el.innerHTML = `<i class="fa-solid ${ic}"></i> <span>${text}</span>`;
}

}

function showToast(msg, type = 'success') {
    const container = document.getElementById('toastContainer');
    if (!container) return;
    const colors = { success: 'bg-green-600', error: 'bg-red-600', info: 'bg-sky', warning: 'bg-amber-500' };
    const icons = { success: 'fa-circle-check', error: 'fa-circle-xmark', info: 'fa-circle-info', warning: 'fa-triangle-exclamation' };
    const toast = document.createElement('div');
    toast.className = `${colors[type] || colors.success} text-white px-4 py-3 rounded-lg shadow-lg font-black text-xs flex items-center gap-2 toast-anim pointer-events-auto max-w-xs`;
    toast.innerHTML = `<i class="fa-solid ${icons[type] || icons.success}"></i> <span>${msg}</span>`;
    container.appendChild(toast);
    setTimeout(() => {
        toast.style.transition = 'all 0.3s';
        toast.style.opacity = '0';
        toast.style.transform = 'translateX(-100%)';
        setTimeout(() => toast.remove(), 300);
    }, 3000);
}

function checkLoginState() {
    const s = sessionStorage.getItem('autocrmUser');
    if (s) {
        const u = JSON.parse(s);
        if (u.roleType === 'customer') { currentUser = u; showCustomerPortal(); }
        else {
            const f = usersDb.find(x => x.id === u.id);
            if (f) { currentUser = { ...f, roleType: 'admin' }; showApp(); }
            else showLogin();
        }
    } else showLogin();
}

async function handleLogin(e) {
    e.preventDefault();
    const u = document.getElementById('loginUser').value.trim();
    const p = document.getElementById('loginPass').value;
    console.log('🔐 محاولة دخول:', u);

    if (currentLoginType === 'admin') {
        // ✅ الأول: دور في Supabase
        let user = null;
        try {
            const { data, error } = await sb.from('users').select('*').eq('username', u).eq('password', p).maybeSingle();
            if (data) {
                user = { id: data.id, username: data.username, password: data.password, name: data.name, phone: data.phone, role: data.role };
                console.log('✅ Supabase user found:', user.name);
            }
        } catch(err) { console.warn('Supabase login error:', err); }

        // ✅ التاني: fallback محلي (DEFAULT_USERS)
        if (!user) {
            user = usersDb.find(x => x.username === u && x.password === p);
            if (user) console.log('✅ Local user found:', user.name);
        }

        if (user) {
            currentUser = { ...user, roleType: 'admin' };
            sessionStorage.setItem('autocrmUser', JSON.stringify(currentUser));
            logActivity('login', `دخول: ${user.name}`);
            showLoading();
            try { await showApp(); } catch(err) { console.error(err); } finally { hideLoading(); }
            showToast(`مرحباً، ${user.name}`);
        } else {
            console.warn('❌ User not found. Available:', usersDb.map(x => x.username));
            showToast('بيانات خاطئة!', 'error');
        }
    } else {
        const cust = appData.customers.find(c => (c.phone === u || c.altPhone === u || c.password === u) && (c.password === p));
        if (cust) {
            currentUser = { ...cust, roleType: 'customer' };
            sessionStorage.setItem('autocrmUser', JSON.stringify(currentUser));
            showCustomerPortal(); showToast(`مرحباً، ${cust.name}`);
        } else showToast('بيانات خاطئة', 'error');
    }
}

async function saveCustomerToDB(customer) {
    try {
        const customerRow = {
            id: customer.id,
            customer_code: customer.password || customer.customerCode || '',
            name: customer.name || '', phone: customer.phone || '',
            alt_phone: customer.altPhone || '', whatsapp: customer.whatsapp || '',
            nat_id: customer.natId || '', job: customer.job || '',
            gov: customer.gov || '', district: customer.district || '',
            address: customer.address || '', channel: customer.channel || '',
            interest: customer.interest || '', status: customer.status || 'مفتوحة',
            notes: customer.notes || '',
            created_by: customer.createdBy || (currentUser ? currentUser.id : ''),
            created_by_name: customer.createdByName || (currentUser ? currentUser.name : ''),
            created_at: customer.createdAt || Date.now(), updated_at: Date.now()
        };
        await sb.from('customers').upsert(customerRow);
        for (const p of (customer.products || [])) {
            if (!p || !p.id) continue;
            await sb.from('products').upsert({
                id: p.id, customer_id: customer.id, vehicle: p.vehicle || '',
                pay_method: p.payMethod || 'كاش', finance_entity: p.financeEntity || '',
                stage: p.stage || '', status: p.status || 'مفتوحة',
                total: Number(p.total || 0), down: Number(p.down || 0),
                monthly: Number(p.monthly || 0), debt: Number(p.debt || 0),
                delivery_date: p.deliveryDate || '', chassis: p.chassis || '',
                motor: p.motor || '', model: p.model || '', notes: p.notes || '',
                created_by: p.createdBy || currentUser.id,
                created_by_name: p.createdByName || currentUser.name,
                created_at: p.createdAt || Date.now()
            });
        }
    } catch(e) { console.warn('Save to DB failed:', e.message); }
}

async function loadCustomersPage(offset = 0, pageSize = 30) {
    const { data, error } = await sb.from('customers').select('*').order('created_at', { ascending: false }).range(offset, offset + pageSize - 1);
    if (error) throw error;
    return {
        customers: (data || []).map(row => ({
            id: row.id, name: row.name, phone: row.phone,
            altPhone: row.alt_phone, whatsapp: row.whatsapp,
            password: row.customer_code, customerCode: row.customer_code,
            natId: row.nat_id, job: row.job, gov: row.gov,
            district: row.district, address: row.address,
            channel: row.channel, interest: row.interest,
            status: row.status, notes: row.notes,
            createdBy: row.created_by, createdByName: row.created_by_name,
            createdAt: row.created_at, products: [], followups: [], _subLoaded: false
        })),
        hasMore: (data || []).length === pageSize
    };
}

async function getCustomerFromDB(id) {
    const cached = loadedCustomers.find(c => c.id === id);
    if (cached && cached._subLoaded && cached.products && cached.products.length) return cached;
    const { data: row, error } = await sb.from('customers').select('*').eq('id', id).single();
    if (error || !row) return null;
    const c = {
        id: row.id, name: row.name, phone: row.phone, altPhone: row.alt_phone,
        whatsapp: row.whatsapp, password: row.customer_code, customerCode: row.customer_code,
        natId: row.nat_id, job: row.job, gov: row.gov, district: row.district,
        address: row.address, channel: row.channel, interest: row.interest,
        status: row.status, notes: row.notes, createdBy: row.created_by,
        createdByName: row.created_by_name, createdAt: row.created_at,
        products: [], followups: [], _subLoaded: true
    };
    try {
        const [pRes, fRes] = await Promise.all([
            sb.from('products').select('*').eq('customer_id', id),
            sb.from('followups').select('*').eq('customer_id', id).order('created_at', { ascending: false })
        ]);
        c.products = (pRes.data || []).map(p => ({
            id: p.id, vehicle: p.vehicle, payMethod: p.pay_method, financeEntity: p.finance_entity,
            stage: p.stage, status: p.status, total: p.total, down: p.down,
            monthly: p.monthly, debt: p.debt, deliveryDate: p.delivery_date,
            chassis: p.chassis, motor: p.motor, model: p.model, notes: p.notes,
            createdBy: p.created_by, createdByName: p.created_by_name, createdAt: p.created_at
        }));
        c.followups = (fRes.data || []).map(f => ({
            id: f.id, date: f.date, time: f.time, reason: f.reason,
            notes: f.notes, byName: f.by_name, byId: f.by_id, createdAt: f.created_at
        }));
    } catch(e) {}
    const idx = loadedCustomers.findIndex(x => x.id === id);
    if (idx >= 0) loadedCustomers[idx] = { ...loadedCustomers[idx], ...c };
    return c;
}

async function detectAndConnect() {
    setCloudStatus('syncing', 'جاري الاتصال...');
    try {
        const { error } = await sb.from('customers').select('id').limit(1);
        if (error) throw error;
        setCloudStatus('online', 'متصل بـ Supabase');
        try {
            const { data: users } = await sb.from('users').select('*');
            if (users && users.length) {
                const dbUsers = users.map(u => ({ id: u.id, username: u.username, password: u.password, name: u.name, phone: u.phone, role: u.role, createdAt: u.created_at }));
                // ادمج مع DEFAULT_USERS (بدون تكرار)
                const existingUsernames = new Set(dbUsers.map(u => u.username));
                DEFAULT_USERS.forEach(du => { if (!existingUsernames.has(du.username)) dbUsers.push(du); });
                usersDb = dbUsers;
            }
        } catch(e) {}
    } catch(e) {
        setCloudStatus('offline', 'وضع محلي');
    }
}

function logActivity(type, details, customerCode = '', customerId = '') {
    try {
        sb.from('activity_log').insert({
            id: 'a' + Date.now() + Math.random().toString(36).slice(2,6),
            type, details, customer_code: customerCode, customer_id: customerId,

        });

        id: 'a' + Date.now(), type, customerCode, customerId,

        timestamp: Date.now(), details: details || ''
    });
    if (appData.activityLog.length > 500) appData.activityLog = appData.activityLog.slice(0, 500);
    saveToLocal();
}

async function loadCustomersInitial() {
    try {
        const cached = localStorage.getItem('autocrmCustomersCache');
        if (cached) {
            const parsed = JSON.parse(cached);
            if (Array.isArray(parsed) && parsed.length) { loadedCustomers = parsed; appData.customers = parsed; updateDashboard(); }
        }
    } catch(e) {}
    getTotalCustomersCount();
    try {
        const page = await Promise.race([loadCustomersPage(0, 30), new Promise((_, r) => setTimeout(() => r(new Error('timeout')), 8000))]);
        const existingIds = new Set(loadedCustomers.map(c => c.id));
        const fresh = page.customers.filter(c => !existingIds.has(c.id));
        loadedCustomers = [...fresh, ...loadedCustomers];
        appData.customers = loadedCustomers;
        lastVisibleOffset = 30;
        hasMoreCustomers = page.hasMore;
        saveCustomersCache();
        updateDashboard(); renderDashboardTasks();
        if (page.hasMore) loadAllRemainingCustomers(); else allCustomersLoaded = true;
    } catch(e) { console.error(e); }
}

async function getTotalCustomersCount() {
    try {
        const { count } = await sb.from('customers').select('*', { count: 'exact', head: true });
        totalCustomersCount = count || 0;
        updateDashboard();
    } catch(e) { totalCustomersCount = Math.max(totalCustomersCount, loadedCustomers.length); }
}

async function loadAllRemainingCustomers() {
    if (backgroundLoadingInProgress) return;
    backgroundLoadingInProgress = true;
    try {
        let offset = lastVisibleOffset, hasMore = hasMoreCustomers, round = 0;
        const existingIds = new Set(loadedCustomers.map(c => c.id));
        while (hasMore) {
            round++;
            const page = await loadCustomersPage(offset, 200);
            const fresh = page.customers.filter(c => !existingIds.has(c.id));
            fresh.forEach(c => existingIds.add(c.id));
            loadedCustomers = loadedCustomers.concat(fresh);
            appData.customers = loadedCustomers;
            offset += 200; hasMore = page.hasMore;
            updateDashboard();
            if (round % 3 === 0) { saveCustomersCache(); await new Promise(r => setTimeout(r, 80)); }
        }
        lastVisibleOffset = offset; hasMoreCustomers = false; allCustomersLoaded = true;
        saveCustomersCache(); updateDashboard();
        showToast(`✅ تم تحميل ${loadedCustomers.length} عميل`);
    } catch(e) {}
    backgroundLoadingInProgress = false;
}

    if (target) { target.classList.remove('hidden'); target.classList.add('block'); }
    document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('active-nav'));
    const active = document.querySelector(`.nav-btn[data-tab="${tabId}"]`);
    if (active) active.classList.add('active-nav');
    closeSidebar();
    if (tabId === 'dashboard') { updateDashboard(); renderDashboardTasks(); }

    if (tabId === 'tasks') renderFullTasks();

    if (tabId === 'activity') { populateActivityFilters(); renderActivityLog(); }
    if (tabId === 'employees') renderEmployeesList();
    if (tabId === 'myactivity') renderMyActivity();
}

function calcDebt() {

    const d = Number(document.getElementById('custDown').value) || 0;
    document.getElementById('custDebt').value = Math.max(0, t - d).toLocaleString() + ' ج';
}
async function saveCustomerData(e) {
    e.preventDefault();
    const id = document.getElementById('custId').value;
    const baseData = {
        id: id || 'c' + Date.now(),
        name: document.getElementById('custName').value.trim(),
        natId: document.getElementById('custNatId').value.trim(),
        job: document.getElementById('custJob').value.trim(),
        phone: document.getElementById('custPhone').value.trim(),
        altPhone: document.getElementById('custAltPhone').value.trim(),
        whatsapp: document.getElementById('custWhatsapp').value.trim(),
        gov: document.getElementById('custGovSelect').value,
        district: document.getElementById('custDistrict').value.trim(),
        address: document.getElementById('custAddress').value.trim(),
        password: document.getElementById('custPassword').value.trim() || generateCustomerCode(),
        channel: document.getElementById('custChannelSelect').value,
        interest: document.getElementById('custInterest').value.trim(),
        status: document.getElementById('custStatus').value,
        notes: document.getElementById('custNotes').value.trim()
    };
    if (id) {
        const existing = await getCustomerFromDB(id);
        if (existing) {
            const updated = { ...existing, ...baseData, _subLoaded: existing._subLoaded };
            await saveCustomerToDB(updated);
            const idx = loadedCustomers.findIndex(c => c.id === id);
            if (idx >= 0) loadedCustomers[idx] = { ...loadedCustomers[idx], ...updated };
        }
        logActivity('edit_customer', `تعديل: ${baseData.name}`);
        showToast('تم التحديث');
    } else {
        const t = Number(document.getElementById('custTotal').value) || 0;
        const d = Number(document.getElementById('custDown').value) || 0;
        const newCustomer = {
            ...baseData,
            products: [{
                id: 'p' + Date.now(), vehicle: baseData.interest || 'غير محدد',
                payMethod: document.getElementById('custPayMethod').value,
                financeEntity: document.getElementById('custFinanceEntity').value,
                stage: '', status: baseData.status, total: t, down: d,
                monthly: Number(document.getElementById('custMonthly').value) || 0,
                deliveryDate: document.getElementById('custDeliveryDate').value || '',
                chassis: '', motor: '', model: '',
                debt: Math.max(0, t - d), notes: '', createdAt: Date.now(),
                createdBy: currentUser.id, createdByName: currentUser.name
            }],
            followups: [], _subLoaded: true,
            createdAt: Date.now(), createdBy: currentUser.id, createdByName: currentUser.name
        };
        await saveCustomerToDB(newCustomer);
        loadedCustomers.unshift(newCustomer);
        totalCustomersCount++;
        logActivity('add_customer', `إضافة: ${baseData.name}`, baseData.password, baseData.id);
        showToast('تم التسجيل - الكود: ' + baseData.password);
    }
    appData.customers = loadedCustomers;
    saveCustomersCache();
    closeCustomerModal();
    updateDashboard();
    if (document.getElementById('tab-search').classList.contains('block')) executeCustomerSearch();
}

function closeCustomerDetail() { document.getElementById('customerDetailModal').classList.add('hidden'); document.getElementById('customerDetailModal').classList.remove('flex'); currentDetailCustomerId = null; }

async function deleteCustomer(id) {
    if (!isManager()) return showToast('للمدير فقط', 'error');
    if (!confirm('حذف العميل؟')) return;
    await sb.from('customers').delete().eq('id', id);
    loadedCustomers = loadedCustomers.filter(x => x.id !== id);
    appData.customers = loadedCustomers;
    if (totalCustomersCount > 0) totalCustomersCount--;
    saveCustomersCache(); updateDashboard();
    closeCustomerDetail();
    if (document.getElementById('tab-search').classList.contains('block')) executeCustomerSearch();
    showToast('تم الحذف');
}

async function openProductModal(custId, prodId) {
    const m = document.getElementById('productModal');
    m.classList.remove('hidden'); m.classList.add('flex');
    document.getElementById('productForm').reset();
    document.getElementById('prodCustId').value = custId;
    document.getElementById('prodId').value = '';
    document.getElementById('prodModalTitle').innerText = 'منتج جديد';
    document.getElementById('prodDebt').value = '0';
    if (prodId) {
        const c = await getCustomerFromDB(custId);
        const p = (c?.products || []).find(x => x.id === prodId);
        if (p) {
            document.getElementById('prodModalTitle').innerText = 'تعديل منتج';
            ['prodVehicle','prodStatus','prodPayMethod','prodFinance','prodStage','prodTotal','prodDown','prodMonthly','prodDeliveryDate','prodChassis','prodMotor','prodModel','prodNotes'].forEach(f => {
                const el = document.getElementById(f);
                if (el) el.value = p[f.replace('prod','').toLowerCase()] || p[f] || '';
            });
            document.getElementById('prodId').value = p.id;
            calcProdDebt();
        }
    }
}
function closeProductModal() { document.getElementById('productModal').classList.add('hidden'); document.getElementById('productModal').classList.remove('flex'); }
function calcProdDebt() {

    const d = Number(document.getElementById('prodDown').value) || 0;

async function saveProductData(e) {
    e.preventDefault();
    const custId = document.getElementById('prodCustId').value;
    const prodId = document.getElementById('prodId').value;
    const t = Number(document.getElementById('prodTotal').value) || 0;
    const d = Number(document.getElementById('prodDown').value) || 0;
    const data = {
        id: prodId || 'p' + Date.now(),
        customer_id: custId,
        vehicle: document.getElementById('prodVehicle').value,
        status: document.getElementById('prodStatus').value,
        pay_method: document.getElementById('prodPayMethod').value,
        finance_entity: document.getElementById('prodFinance').value,
        stage: document.getElementById('prodStage').value,
        total: t, down: d,
        monthly: Number(document.getElementById('prodMonthly').value) || 0,
        delivery_date: document.getElementById('prodDeliveryDate').value || '',
        chassis: document.getElementById('prodChassis').value.trim(),
        motor: document.getElementById('prodMotor').value.trim(),
        model: document.getElementById('prodModel').value.trim(),
        debt: Math.max(0, t - d),
        notes: document.getElementById('prodNotes').value.trim(),
        created_at: Date.now(),
        created_by: currentUser.id, created_by_name: currentUser.name
    };
    await sb.from('products').upsert(data);
    const idx = loadedCustomers.findIndex(c => c.id === custId);
    if (idx >= 0) {
        const prods = loadedCustomers[idx].products || [];
        const pIdx = prods.findIndex(p => p.id === data.id);
        const prodObj = { id: data.id, vehicle: data.vehicle, payMethod: data.pay_method, financeEntity: data.finance_entity, stage: data.stage, status: data.status, total: data.total, down: data.down, monthly: data.monthly, debt: data.debt, deliveryDate: data.delivery_date, chassis: data.chassis, motor: data.motor, model: data.model, notes: data.notes, createdAt: data.created_at };
        if (pIdx >= 0) prods[pIdx] = prodObj; else prods.push(prodObj);
        loadedCustomers[idx].products = prods;
    }
    const c = loadedCustomers.find(x => x.id === custId);
    logActivity(prodId ? 'edit_product' : 'add_product', `${prodId ? 'تعديل' : 'إضافة'} منتج لـ ${c?.name || ''}`);
    closeProductModal();
    if (currentDetailCustomerId === custId) openCustomerDetail(custId);
    showToast('تم الحفظ');
}
async function deleteProduct(custId, prodId) {
    if (!isManager()) return showToast('للمدير فقط', 'error');
    if (!confirm('حذف المنتج؟')) return;
    await sb.from('products').delete().eq('id', prodId);
    const idx = loadedCustomers.findIndex(c => c.id === custId);
    if (idx >= 0) loadedCustomers[idx].products = (loadedCustomers[idx].products || []).filter(p => p.id !== prodId);
    if (currentDetailCustomerId === custId) openCustomerDetail(custId);
    showToast('تم الحذف');
}

async function executeCustomerSearch() {
    const q = document.getElementById('mainSearchPhone').value.trim();
    const area = document.getElementById('searchResultsArea');
    const loadMoreWrap = document.getElementById('loadMoreWrap');
    if (!q) {
        if (loadedCustomers.length === 0) {
            area.innerHTML = '<div class="panel p-6 text-center"><i class="fa-solid fa-users text-gray-300 text-4xl mb-2"></i><div class="text-gray-400 font-bold text-sm">لا يوجد عملاء</div></div>';
            loadMoreWrap.classList.add('hidden'); return;
        }
        const toShow = loadedCustomers.slice(0, 50);
        const status = backgroundLoadingInProgress && !allCustomersLoaded ? ` <span class="text-sky animate-pulse text-xs">⏳ ${loadedCustomers.length}/${totalCustomersCount}</span>` : '';
        area.innerHTML = `<div class="text-xs font-black text-gray-500 mb-2"><i class="fa-solid fa-users text-sky"></i> ${allCustomersLoaded ? `كل العملاء (${loadedCustomers.length})` : `محمل: ${loadedCustomers.length}/${totalCustomersCount}`}${status}</div>` + toShow.map(c => renderCustomerCard(c)).join('');
        if (loadedCustomers.length > 50) loadMoreWrap.classList.remove('hidden'); else loadMoreWrap.classList.add('hidden');
        return;
    }
    area.innerHTML = '<div class="text-center py-8"><i class="fa-solid fa-spinner fa-spin text-2xl text-sky"></i></div>';
    loadMoreWrap.classList.add('hidden');
    try {
        const qLower = q.toLowerCase();
        const results = loadedCustomers.filter(c =>
            (c.name || '').toLowerCase().includes(qLower) ||
            (c.phone || '').includes(q) ||
            (c.altPhone || '').includes(q) ||
            (c.whatsapp || '').includes(q) ||
            (c.customerCode || '').toUpperCase().includes(q.toUpperCase()) ||
            (c.natId || '').includes(q)
        );
        if (results.length === 0) {
            area.innerHTML = '<div class="panel p-6 text-center"><i class="fa-solid fa-magnifying-glass text-gray-300 text-4xl mb-2"></i><div class="text-gray-500 font-black text-sm">لا توجد نتائج</div></div>';
            return;
        }
        area.innerHTML = `<div class="text-xs font-black text-gray-500 mb-2"><i class="fa-solid fa-circle-check text-green-600"></i> نتائج (${results.length})</div>` + results.slice(0, 100).map(c => renderCustomerCard(c)).join('');
    } catch(e) { area.innerHTML = `<div class="panel p-6 text-center text-red-500 text-sm">خطأ</div>`; }
}

function renderCustomerCard(c) {
    const products = c.products || [];
    const lastProduct = products.length ? products[products.length - 1] : {};
    const status = lastProduct.status || c.status || 'مفتوحة';
    const map = { 'منفذة': 'badge-green', 'جاري التقسيط': 'badge-yellow', 'مفتوحة': 'badge-blue', 'مفقودة': 'badge-red' };
    const statusCls = map[status] || 'badge-gray';
    const totalDebt = products.reduce((sum, p) => sum + (Number(p.debt) || 0), 0);
    const firstLetter = (c.name || '?').charAt(0).toUpperCase();
    return `<div class="panel p-3 mb-2 task-clickable" onclick="openCustomerDetail('${c.id}')">
        <div class="flex items-start gap-3">
            <div class="w-10 h-10 rounded-lg bg-gradient-to-br from-navy-dark to-sky flex items-center justify-center text-white font-black text-sm shrink-0">${firstLetter}</div>
            <div class="flex-1 min-w-0">
                <div class="flex flex-wrap items-center gap-1.5 mb-1">
                    <b class="text-navy-dark text-sm">${c.name || '-'}</b>
                    <span class="badge ${statusCls}">${status}</span>
                    ${c.customerCode ? `<span class="badge badge-blue tracking-widest text-[10px]">${c.customerCode}</span>` : ''}
                </div>
                <div class="text-[11px] text-gray-500 font-bold flex flex-wrap gap-x-3 gap-y-1">
                    ${c.phone ? `<span><i class="fa-solid fa-phone text-sky"></i> ${c.phone}</span>` : ''}
                    ${c.gov ? `<span><i class="fa-solid fa-location-dot text-sky"></i> ${c.gov}</span>` : ''}
                    ${lastProduct.vehicle ? `<span><i class="fa-solid fa-car text-sky"></i> ${lastProduct.vehicle}</span>` : ''}
                </div>
            </div>
            <div class="text-left shrink-0">
                ${totalDebt > 0 ? `<div class="text-[10px] text-amber-700 font-black">${totalDebt.toLocaleString()} ج</div>` : ''}
            </div>
        </div>
    </div>`;
}

async function loadMoreCustomers() {
    const area = document.getElementById('searchResultsArea');
    const currentShown = area.querySelectorAll('.panel.task-clickable').length;
    const toShow = loadedCustomers.slice(currentShown, currentShown + 50);
    if (toShow.length > 0) area.insertAdjacentHTML('beforeend', toShow.map(c => renderCustomerCard(c)).join(''));
    if (currentShown + toShow.length >= loadedCustomers.length && allCustomersLoaded) { document.getElementById('loadMoreWrap').classList.add('hidden'); showToast('وصلت لآخر عميل'); }
}

function updateDashboard() {
    const customers = appData.customers || [];
    const displayCount = totalCustomersCount > customers.length ? totalCustomersCount : customers.length;
    document.getElementById('kpiTotalCustomers').innerText = displayCount.toLocaleString();
    let salesDone = 0, totalDebt = 0;
    customers.forEach(c => { (c.products || []).forEach(p => { if ((p.status || '').includes('منفذة')) salesDone++; totalDebt += Number(p.debt) || 0; }); });
    document.getElementById('kpiSalesDone').innerText = salesDone;
    document.getElementById('kpiTotalDebt').innerText = totalDebt.toLocaleString();
    const today = new Date().toISOString().split('T')[0];
    const allPending = (appData.tasks || []).filter(t => !t.completed);
    const todayTasks = allPending.filter(t => t.date === today);
    document.getElementById('kpiTodayTasks').innerText = todayTasks.length;
    document.getElementById('pendingTasksBadge').innerText = allPending.length;
    const navCount = document.getElementById('navTasksCount');
    if (allPending.length > 0) { navCount.innerText = allPending.length; navCount.classList.remove('hidden'); } else navCount.classList.add('hidden');
}

function renderDashboardTasks() {
    const container = document.getElementById('dashboardTasksList');
    if (!container) return;
    const today = new Date().toISOString().split('T')[0];
    const tasks = (appData.tasks || []).filter(t => !t.completed).sort((a, b) => (a.date || '').localeCompare(b.date || '')).slice(0, 6);
    if (tasks.length === 0) { container.innerHTML = '<div class="text-center text-gray-400 py-6 font-bold text-sm">لا توجد مهام</div>'; return; }
    container.innerHTML = tasks.map(t => {
        const cust = loadedCustomers.find(c => c.id === t.customerId);
        const isToday = t.date === today; const isOverdue = t.date < today;

function filterCustomers(customers, f) {
    return customers.filter(c => {
        if (f.gov && (c.gov || '') !== f.gov) return false;
        if (f.district && !(c.district || '').includes(f.district)) return false;
        if (f.channel && (c.channel || '') !== f.channel) return false;
        const products = c.products || [];
        if (f.status && !products.some(p => (p.status || '').includes(f.status)) && !(c.status || '').includes(f.status)) return false;
        if (f.payMethod && !products.some(p => p.payMethod === f.payMethod)) return false;
        if (f.vehicle && !products.some(p => p.vehicle === f.vehicle)) return false;
        if (f.finance && !products.some(p => p.financeEntity === f.finance)) return false;
        return true;
    });
}

async function handleLocalFileImport(event) {
    const file = event.target.files[0];
    if (!file) return;
    const ext = file.name.split('.').pop().toLowerCase();
    if (!['xlsx','xls','csv'].includes(ext)) { showToast('صيغة غير مدعومة', 'error'); event.target.value = ''; return; }
    await processImportFile(file);
    event.target.value = '';
}
async function processImportFile(file) {
    if (!file) return;
    showLoading();
    showToast('جاري القراءة: ' + file.name, 'info');
    const reader = new FileReader();
    reader.onload = async (e) => {
        try {
            const data = new Uint8Array(e.target.result);
            const wb = XLSX.read(data, { type: 'array' });
            const sheet = wb.Sheets[wb.SheetNames[0]];
            const rawRows = XLSX.utils.sheet_to_json(sheet, { header: 1, defval: '', blankrows: false });
            if (!rawRows.length) { showToast('الملف فاضي', 'error'); hideLoading(); return; }
            const knownKeywords = ['الاسم','اسم','name','Name','الهاتف','هاتف','phone','Phone','الموبايل','موبايل','رقم'];
            let headerRowIndex = -1, bestScore = 0;
            for (let i = 0; i < Math.min(10, rawRows.length); i++) {
                const row = rawRows[i]; if (!Array.isArray(row)) continue;
                let score = 0;
                row.forEach(cell => { const val = String(cell || '').trim().toLowerCase(); if (!val) return; if (knownKeywords.some(k => val.includes(k.toLowerCase()))) score += 3; else if (val.length > 0 && val.length < 30) score += 1; });
                if (score > bestScore) { bestScore = score; headerRowIndex = i; }
            }
            if (headerRowIndex === -1 || bestScore < 3) { alert('⚠️ مفيش صف عناوين واضح!'); hideLoading(); return; }
            const headers = rawRows[headerRowIndex].map(h => String(h || '').trim());
            const rows = [];
            for (let i = headerRowIndex + 1; i < rawRows.length; i++) {
                const rawRow = rawRows[i]; if (!Array.isArray(rawRow)) continue;
                const obj = {}; let hasData = false;
                headers.forEach((h, idx) => { if (!h) return; obj[h] = rawRow[idx] ?? ''; if (String(rawRow[idx] || '').trim()) hasData = true; });
                if (hasData) rows.push(obj);
            }
            if (!rows.length) { showToast('مفيش بيانات', 'error'); hideLoading(); return; }
            if (!confirm(`استيراد ${rows.length} صف؟`)) { hideLoading(); return; }
            const getVal = (row, keys) => { for (const k of keys) if (row[k] !== undefined && String(row[k]).trim() !== '') return String(row[k]).trim(); const rowKeys = Object.keys(row); for (const k of keys) { const found = rowKeys.find(rk => rk.toLowerCase().includes(k.toLowerCase())); if (found && String(row[found] || '').trim()) return String(row[found]).trim(); } return ''; };
            let imported = 0, skipped = 0, failed = 0;
            const importedCustomers = [];
            for (const row of rows) {
                try {
                    const name = getVal(row, ['الاسم','الاسم الكامل','اسم العميل','اسم','name','Name']);
                    const phone = getVal(row, ['الهاتف','رقم الهاتف','الهاتف الأساسي','الموبايل','phone','Phone','mobile']);
                    if (!name || !phone) { skipped++; continue; }
                    let code = getVal(row, ['كود العميل','كود','code','customerCode']) || generateCustomerCode();
                    const total = Number(getVal(row, ['الإجمالي','total']) || 0) || 0;
                    const down = Number(getVal(row, ['المقدم','down']) || 0) || 0;
                    const vehicle = getVal(row, ['المركبة','المركبة المطلوبة','vehicle','السيارة']) || 'غير محدد';
                    const financeEntity = getVal(row, ['جهة التقسيط','الجهة','finance']);

(function initDragDrop() {
    let dragCounter = 0;
    const overlay = document.getElementById('dropOverlay');
    if (!overlay) return;
    const isFileDrag = (e) => { if (!e.dataTransfer) return false; const types = e.dataTransfer.types; return types && Array.from(types).includes('Files'); };
    window.addEventListener('dragenter', (e) => { if (!isFileDrag(e)) return; e.preventDefault(); dragCounter++; if (!currentUser || currentUser.role !== 'manager') return; const ml = document.getElementById('mainLayout'); if (!ml || ml.classList.contains('hidden')) return; overlay.classList.add('active'); });
    window.addEventListener('dragover', (e) => { if (!isFileDrag(e)) return; e.preventDefault(); });
    window.addEventListener('dragleave', (e) => { if (!isFileDrag(e)) return; e.preventDefault(); dragCounter--; if (dragCounter <= 0) { dragCounter = 0; overlay.classList.remove('active'); } });
    window.addEventListener('drop', async (e) => {
        if (!isFileDrag(e)) return; e.preventDefault(); dragCounter = 0; overlay.classList.remove('active');
        if (!currentUser || currentUser.role !== 'manager') { showToast('للمدير فقط', 'error'); return; }
        const files = e.dataTransfer.files; if (!files || files.length === 0) return;
        const file = files[0]; const ext = file.name.split('.').pop().toLowerCase();
        if (!['xlsx','xls','csv'].includes(ext)) { showToast('صيغة غير مدعومة', 'error'); return; }
        await processImportFile(file);
    });
    window.addEventListener('dragover', (e) => { if (isFileDrag(e)) e.preventDefault(); }, false);
    window.addEventListener('drop', (e) => { if (isFileDrag(e)) e.preventDefault(); }, false);
})();
